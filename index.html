import os import re import json import docx folder_path = r"C:\Users\Manjith\Downloads\attachments" output_file = os.path.join(folder_path, "vsg_bills_backup.json") def clean_text(text): return re.sub(r'\s+', ' ', text).strip() if text else "" def parse_num(val_str, default=0.0): if not val_str: return default m = re.search(r'\d+(?:\.\d+)?', val_str.replace(',', '')) if m: try: return float(m.group(0)) except: return default return default def calculate_cycle(date_str): match = re.search(r'(\d{4})', date_str) if match: year = int(match.group(1)) next_yr_short = str(year + 1)[-2:] return str(year) + "-" + next_yr_short return "2026-27" documents = [] customers_dict = {} for filename in os.listdir(folder_path): if filename.endswith(".docx") and not filename.startswith("~$"): file_full_path = os.path.join(folder_path, filename) try: doc = docx.Document(file_full_path) full_text = "\n".join([p.text for p in doc.paragraphs]) for t in doc.tables: for row in t.rows: full_text += "\n" + " ".join([c.text for c in row.cells]) # Extract invoice number inv_match = re.search(r'(?:INVOICE\s*NO|Invoice\s*No|Inv\s*No)[\s.:]*([A-Za-z0-9\-_/]+)', full_text, re.IGNORECASE) if inv_match: inv_no = inv_match.group(1).strip() else: num_prefix = re.match(r'^(\d+)', filename) inv_no = num_prefix.group(1) if num_prefix else filename.replace(".docx", "") # Extract Date date_match = re.search(r'(?:Date|DATE)[\s.:]*([0-9]{2}[-/][0-9]{2}[-/][0-9]{4}|[0-9]{4}[-/][0-9]{2}[-/][0-9]{2})', full_text) raw_date = date_match.group(1).strip() if date_match else "2026-04-01" date_parts = re.split(r'[-/]', raw_date) if len(date_parts) == 3: if len(date_parts[0]) == 4: iso_date = str(date_parts[0]) + "-" + str(date_parts[1]).zfill(2) + "-" + str(date_parts[2]).zfill(2) else: iso_date = str(date_parts[2]) + "-" + str(date_parts[1]).zfill(2) + "-" + str(date_parts[0]).zfill(2) else: iso_date = "2026-04-01" # Extract Customer Name cust_name = "" name_match = re.search(r'(?:Name|M/s|To)\s*[:.]*\s*([^\n\r,]+)', full_text, re.IGNORECASE) if name_match: candidate = clean_text(name_match.group(1)) if "vsg industries" not in candidate.lower(): cust_name = candidate if not cust_name: cleaned_fn = re.sub(r'^\d+\s*', '', filename.replace(".docx", "")).strip() cust_name = cleaned_fn if cleaned_fn else "Direct Customer" # Extract GSTIN & Phone gstin_match = re.search(r'33[A-Z]{5}[0-9]{4}[A-Z]{1}[1-9A-Z]{1}Z[0-9A-Z]{1}', full_text) gstin = gstin_match.group(0) if gstin_match and gstin_match.group(0) != "33AJVPG9441P1ZE" else "" phone_match = re.search(r'(?:Contact|Mob|Ph)[\s.:]*([6-9][0-9]{9})', full_text) phone = phone_match.group(1) if phone_match and phone_match.group(1) != "9942947305" else "" if cust_name not in customers_dict: cust_id = 1000 + len(customers_dict) + 1 customers_dict[cust_name] = { "id": cust_id, "name": cust_name, "keyword": cust_name[:6].upper(), "gstin": gstin, "phone": phone, "address": "Coimbatore" } # Locate ONLY the actual item table items = [] subtotal = 0.0 for table in doc.tables: header_text = " ".join([clean_text(c.text).lower() for c in table.rows[0].cells]) if table.rows else "" # Check if this table is the goods breakdown table if any(k in header_text for k in ["description", "particulars", "goods", "qts", "rate"]): for row in table.rows[1:]: cells = [clean_text(c.text) for c in row.cells] if len(cells) >= 3: desc = cells[1] if len(cells) > 3 else cells[0] qty_str = cells[2] if len(cells) > 3 else cells[1] rate_str = cells[3] if len(cells) > 3 else cells[2] qty_val = parse_num(qty_str, 1.0) rate_val = parse_num(rate_str, 0.0) # Filter out headers, totals, bank info, or phone numbers invalid_terms = ["total", "sub total", "grand total", "cgst", "sgst", "bank", "cell", "industries", "signature", "receiver", "layout"] if desc and not any(term in desc.lower() for term in invalid_terms): if rate_val < 5000000 and qty_val < 10000: # Exclude phone number values if rate_val > 0: items.append({ "desc": desc, "qty": str(int(qty_val)) if qty_val.is_integer() else str(qty_val), "rate": rate_val }) subtotal += (qty_val * rate_val) # Fallback if specific table wasn't matched if not items: items.append({"desc": "General Lathe & Fabrication Works", "qty": "1", "rate": 1000.0}) subtotal = 1000.0 sgst = round(subtotal * 0.09, 2) cgst = round(subtotal * 0.09, 2) raw_grand = subtotal + sgst + cgst grand_total = round(raw_grand) round_off = round(grand_total - raw_grand, 2) formatted_inv_no = str(inv_no).zfill(3) if str(inv_no).isdigit() else str(inv_no) documents.append({ "type": "INVOICE", "number": formatted_inv_no, "cycle": calculate_cycle(iso_date), "date": iso_date, "vehicle": "", "customer": cust_name, "gstin": gstin, "phone": phone, "address": "Coimbatore", "items": items, "subtotal": subtotal, "sgst": sgst, "cgst": cgst, "roundOff": round_off, "total": grand_total, "status": "PAID" }) print(f"Cleaned: {filename} -> Inv #{formatted_inv_no} (Total: Rs.{grand_total})") except Exception as e: print(f"Error reading {filename}: {e}") backup_json = { "version": "2.0", "exportDate": "2026-08-28T00:00:00.000Z", "profile": { "name": "VSG Industries", "tagline": "ALL KINDS OF AUTOMOBILE LATHE, FABRICATION & WELDING WORKS", "address1": "55,Tank Road Nataraj Gounder Layout,", "address2": "Sivanandha Colony,Coimbatore-641028", "contact": "9942947305", "gstin": "33AJVPG9441P1ZE", "bankName": "Indian Overseas Bank", "bankAcc": "170902000001290", "bankIFSC": "IOBA0001709", "bankBranch": "Rathinapuri,CBE-27" }, "customers": list(customers_dict.values()), "documents": sorted(documents, key=lambda x: str(x["number"])), "payments": [], "userCycles": list(set([d["cycle"] for d in documents])) } with open(output_file, "w", encoding="utf-8") as f: json.dump(backup_json, f, indent=2, ensure_ascii=False) print(f"\nCompleted! Generated clean backup with {len(documents)} bills.")
